# TP1 — Partie 4 : agrégations

# Exercice 4.1 — Offres et salaire moyen par ville
GET offres/_search
{
  "size": 0,
  "aggs": {
    "par_ville": {
      "terms": {
        "field": "ville",
        "size": 12,
        "order": {
          "salaire_moyen": "desc"
        }
      },
      "aggs": {
        "salaire_moyen": {
          "avg": {
            "field": "salaire_min"
          }
        }
      }
    }
  }
}

# Exercice 4.3 — Tranches de salaire et statistiques d'expérience
GET offres/_search
{
  "size": 0,
  "aggs": {
    "tranches_salaire": {
      "range": {
        "field": "salaire_min",
        "ranges": [
          { "key": "< 40 k", "to": 40000 },
          { "key": "40-55 k", "from": 40000, "to": 55000 },
          { "key": ">= 55 k", "from": 55000 }
        ]
      }
    },
    "experience": {
      "stats": {
        "field": "experience_annees"
      }
    }
  }
}
